cmake_minimum_required(VERSION 3.16)
project(p2pro_protocol LANGUAGES CXX)
add_library(p2pro_protocol STATIC native/p2pro_protocol.cpp native/p2pro_camera.cpp native/radiometry.cpp native/measurements.cpp)
target_include_directories(p2pro_protocol PUBLIC native)
target_compile_features(p2pro_protocol PUBLIC cxx_std_17)
target_compile_options(p2pro_protocol PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_protocol native/test_protocol.cpp)
    target_link_libraries(test_protocol PRIVATE p2pro_protocol)
    add_test(NAME captured_frame_and_packets COMMAND test_protocol
        ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/captured-desktop.yuyv)
    add_executable(test_radiometry native/test_radiometry.cpp)
    target_link_libraries(test_radiometry PRIVATE p2pro_protocol)
    add_test(NAME band_radiometry COMMAND test_radiometry)
    add_executable(test_orientation native/test_orientation.cpp)
    target_link_libraries(test_orientation PRIVATE p2pro_protocol)
    add_test(NAME orientation COMMAND test_orientation)
    add_executable(test_measurements native/test_measurements.cpp)
    target_link_libraries(test_measurements PRIVATE p2pro_protocol)
    add_test(NAME sensor_measurements COMMAND test_measurements)
endif()
